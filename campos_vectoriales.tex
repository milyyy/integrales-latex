\subsection*{15.- Campo vectorial del Ejercicio 15}
\begin{minipage}[t]{0.5\textwidth}
\textbf{Procedimiento:}
\[
y' + ty = y
\]
\[
y' + (t-1)y = 0
\]
\[
\frac{dy}{dt} = (1-t)y
\]
\[
\frac{dy}{y}=(1-t)dt
\]
\[
\int\frac{dy}{y}=\int(1-t)dt
\]
\[
\ln|y|=t-\frac{t^2}{2}+C
\]
\[
|y|=Ce^{t-\frac{t^2}{2}}
\]
Con $y(1)=3$:
\[
3=Ce^{1-\frac{1}{2}}=Ce^{1/2}
\]
\[
C=3e^{-1/2}
\]
\[
|y|=3e^{t-\frac{t^2}{2}-\frac12}
\]
\end{minipage}
